cmake_minimum_required(VERSION 3.14)
if(NOT DEFINED CMAKE_TOOLCHAIN_FILE AND DEFINED ENV{VCPKG_ROOT})
 set(CMAKE_TOOLCHAIN_FILE "$ENV{VCPKG_ROOT}/scripts/buildsystems/vcpkg.cmake" CACHE FILEPATH "Toolchain")
endif()
project(LighthouseExtended VERSION 0.1.0 LANGUAGES C CXX)
option(LIGHTHOUSE_EXTENDED_BUILD_TESTING "Build Extended tests" OFF)
option(LIGHTHOUSE_EXTENDED_BUILD_EXAMPLES "Build Extended examples" OFF)
if(NOT TARGET Lighthouse::Engine)
 add_subdirectory(extern/Lighthouse-Engine)
endif()
include(extern/Lighthouse-Engine/cmake/EmbedResources.cmake)
lighthouse_embed_resources(LighthouseExtendedEmbedded "Lighthouse::ExtendedEmbedded"
    "Lighthouse/Extended/EmbeddedResources.hpp" "${CMAKE_CURRENT_SOURCE_DIR}/res")
add_library(Lighthouse::ExtendedEmbedded ALIAS LighthouseExtendedEmbedded)
target_include_directories(LighthouseExtendedEmbedded PUBLIC "$<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>")
find_package(EnTT CONFIG REQUIRED)
find_package(nlohmann_json CONFIG REQUIRED)
file(GLOB_RECURSE extended_sources CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/src/*.cpp")
add_library(LighthouseExtended STATIC ${extended_sources})
add_library(Lighthouse::Extended ALIAS LighthouseExtended)
target_include_directories(LighthouseExtended PUBLIC "$<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>")
target_include_directories(LighthouseExtended SYSTEM PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/extern/stb")
target_compile_features(LighthouseExtended PUBLIC cxx_std_23)
target_link_libraries(LighthouseExtended PUBLIC Lighthouse::Engine Lighthouse::ExtendedEmbedded EnTT::EnTT PRIVATE nlohmann_json::nlohmann_json)
if(LIGHTHOUSE_IMGUI)
 find_package(graphviz CONFIG REQUIRED)
 find_package(unofficial-imgui-node-editor CONFIG REQUIRED)
 # Graphviz does not export CMake targets for its shared layout plugins.
 get_target_property(_gvc_release graphviz::gvc IMPORTED_LOCATION_RELEASE)
 get_target_property(_gvc_debug graphviz::gvc IMPORTED_LOCATION_DEBUG)
 get_filename_component(_gvc_release_dir "${_gvc_release}" DIRECTORY)
 get_filename_component(_gvc_debug_dir "${_gvc_debug}" DIRECTORY)
 foreach(_layout dot neato)
  string(TOUPPER "${_layout}" _layout_upper)
  find_library(LIGHTHOUSE_${_layout_upper}_RELEASE NAMES gvplugin_${_layout}_layout HINTS "${_gvc_release_dir}" PATH_SUFFIXES graphviz NO_DEFAULT_PATH REQUIRED)
  find_library(LIGHTHOUSE_${_layout_upper}_DEBUG NAMES gvplugin_${_layout}_layout HINTS "${_gvc_debug_dir}" PATH_SUFFIXES graphviz NO_DEFAULT_PATH REQUIRED)
  target_link_libraries(LighthouseExtended PRIVATE
      debug "${LIGHTHOUSE_${_layout_upper}_DEBUG}" optimized "${LIGHTHOUSE_${_layout_upper}_RELEASE}")
 endforeach()
 target_link_libraries(LighthouseExtended PRIVATE graphviz::gvc graphviz::cgraph unofficial::imgui-node-editor::imgui-node-editor)
endif()

if(LIGHTHOUSE_EXTENDED_BUILD_TESTING)
 enable_testing()
 add_subdirectory(tests)
endif()
if(LIGHTHOUSE_EXTENDED_BUILD_EXAMPLES)
 add_subdirectory(Examples)
endif()
